# PMocker M2 内核阶段实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 PMocker 内核三件套——EAV 引擎（元表+业务表+专用表+CRUD）、三维权限判定器、声明式工作流引擎，并通过 API 验证可加载 schema、创建动态字段实体、跑通工作流。

**Architecture:** 依赖倒置。`pkg/pmocker/` 定义纯类型和接口（不依赖 gva/gorm），`gva/server/model/pmocker/` 定义 GORM model，`gva/server/service/pmocker/` 实现 EAV 引擎。gva 通过 go.work require pkg，pkg 不反向依赖 gva。表注册走 gva 的 `bizModel()` 扩展（在 gva/server/initialize/ 下新增文件，不改 subtree 原文件）。

**Tech Stack:** Go 1.24 + GORM + gin-vue-admin v3 + SQLite

## Global Constraints

- gva module path: `github.com/flipped-aurora/gin-vue-admin/server`
- pkg module path: `github.com/pmocker-io/pmocker/pkg`
- 不修改 gva subtree 原文件（subtree pull 时不冲突）
- PMocker 代码放 `gva/server/model/pmocker/`、`gva/server/service/pmocker/`、`gva/server/api/v1/pmocker/`、`gva/server/router/pmocker/`
- 表注册通过新建 `gva/server/initialize/pmocker_init.go`（不改 gorm_biz.go），在 `RegisterTables` 末尾通过 init 钩子调用
- GORM model 嵌入 `global.GVA_MODEL`（含 ID/CreatedAt/UpdatedAt/DeletedAt）
- SQLite 不支持 LTREE，WBS/EPS 路径用字符串（如 `1.3.2`）
- EAV 元表由 schema 层灌入（M2 先用 Go 代码硬编码灌入，M4 改为从 .pmi 镜像读取）

## gva 关键模式参考

**GVA_MODEL**（`gva/server/global/model.go`）：
```go

    ID        uint           `gorm:"primarykey" json:"ID"`

// Package plugin 定义 PMocker 插件扩展接口。
package plugin

import "context"

// PMockerPlugin 扩展 gva plugin.Plugin 接口（server/utils/plugin/v2）。
// gva 原生插件不实现此接口也能正常运行（兼容）。
type PMockerPlugin interface {
    // InitPMocker 在 PMSystem 启动时由内核调用。
    // 用于：灌入 schema/seed/menu/工作流到 EAV 元表 + gva 系统表。
    InitPMocker(ctx context.Context) error
}

// InitAllPMockerPlugins 扫描已注册插件，对实现 PMockerPlugin 的调用 InitPMocker。
// gva 原生插件（未实现 PMockerPlugin）跳过，不影响运行。
func InitAllPMockerPlugins(ctx context.Context, plugins []interface{}) []error {
    var errs []error
    for _, p := range plugins {
        if pmockerPlugin, ok := p.(PMockerPlugin); ok {
            if err := pmockerPlugin.InitPMocker(ctx); err != nil {
                errs = append(errs, err)
            }
        }
    }
    return errs
}
```

        attr.ValInt = &v
    case float64:
        attr.ValDecimal = &v
    case bool:
        attr.ValBool = &v
    default:
        // 复杂类型序列化为 JSON
        attr.ValJSON = nil // TODO: json.Marshal
    }
}

    if attr.ValInt != nil {
        return *attr.ValInt

    if attr.ValDecimal != nil {
        return *attr.ValDecimal
    }
    if attr.ValBool != nil {
        return *attr.ValBool
    }
    if attr.ValJSON != nil {
        return *attr.ValJSON
    }
    return nil
}

import (
    "context"
    "encoding/json"
    "fmt"

    "github.com/flipped-aurora/gin-vue-admin/server/global"
    "github.com/flipped-aurora/gin-vue-admin/server/model/pmocker"
    "github.com/pmocker-io/pmocker/pkg/pmocker/workflow"
)

type WorkflowService struct{}

// LoadDefinition 加载工作流定义到 DB
func (s *WorkflowService) LoadDefinition(ctx context.Context, def workflow.Definition) error {
    defJSON, _ := json.Marshal(def)
    return global.GVA_DB.WithContext(ctx).Where("code = ?", def.Code).
        Assign(pmocker.PMWorkflowDef{
            Name:           def.Name,
            EntityType:     def.EntityType,
            Trigger:        def.Trigger,
            DefinitionJSON: string(defJSON),
        }).FirstOrCreate(&pmocker.PMWorkflowDef{Code: def.Code}).Error
}

// Start 启动工作流实例
func (s *WorkflowService) Start(ctx context.Context, entityID uint, defCode string) (uint, error) {
    // 加载定义
    var def pmocker.PMWorkflowDef
    if err := global.GVA_DB.WithContext(ctx).Where("code = ?", defCode).First(&def).Error; err != nil {
        return 0, fmt.Errorf("workflow def %s not found: %w", defCode, err)
    }
    var wf workflow.Definition
    if err := json.Unmarshal([]byte(def.DefinitionJSON), &wf); err != nil {
        return 0, fmt.Errorf("invalid workflow definition: %w", err)
    }
    // 找 start 节点
    var startNode *workflow.Node
    for i := range wf.Nodes {
        if wf.Nodes[i].Type == workflow.NodeStart {
            startNode = &wf.Nodes[i]
            break
        }
    }
    if startNode == nil {
        return 0, fmt.Errorf("no start node in workflow %s", defCode)
    }
    // 创建实例
    inst := pmocker.PMWorkflowInstance{
        EntityID:     entityID,
        WorkflowCode: defCode,
        CurrentNode:  startNode.Code,
        Status:       "running",
    }
    if err := global.GVA_DB.WithContext(ctx).Create(&inst).Error; err != nil {
        return 0, err
    }
    return inst.ID, nil
}

// Execute 执行转移
func (s *WorkflowService) Execute(ctx context.Context, instanceID uint, action string, userID uint) error {
    var inst pmocker.PMWorkflowInstance
    if err := global.GVA_DB.WithContext(ctx).First(&inst, instanceID).Error; err != nil {
        return err
    }
    if inst.Status != "running" {
        return fmt.Errorf("workflow instance %d is not running", instanceID)
    }
    // 加载定义
    var def pmocker.PMWorkflowDef
    if err := global.GVA_DB.WithContext(ctx).Where("code = ?", inst.WorkflowCode).First(&def).Error; err != nil {
        return err
    }
    var wf workflow.Definition
    if err := json.Unmarshal([]byte(def.DefinitionJSON), &wf); err != nil {
        return err
    }
    // 查找匹配的转移
    for _, t := range wf.Transitions {
        if t.From == inst.CurrentNode && (t.On == "" || t.On == action) {
            // 更新实例
            update := map[string]interface{}{"current_node": t.To}
            if t.Status != "" {
                update["status"] = t.Status
            }
            // 检查目标是否是 end 节点
            for _, n := range wf.Nodes {
                if n.Code == t.To && n.Type == workflow.NodeEnd {
                    update["status"] = "completed"
                    break
                }
            }
            return global.GVA_DB.WithContext(ctx).Model(&pmocker.PMWorkflowInstance{}).
                Where("id = ?", instanceID).Updates(update).Error
        }
    }
    return fmt.Errorf("no transition from %s on action %s", inst.CurrentNode, action)
}

// GetCurrentNode 获取当前节点
func (s *WorkflowService) GetCurrentNode(ctx context.Context, instanceID uint) (*workflow.Node, error) {
    var inst pmocker.PMWorkflowInstance
    if err := global.GVA_DB.WithContext(ctx).First(&inst, instanceID).Error; err != nil {
        return nil, err
    }
    var def pmocker.PMWorkflowDef
    if err := global.GVA_DB.WithContext(ctx).Where("code = ?", inst.WorkflowCode).First(&def).Error; err != nil {
        return nil, err
    }
    var wf workflow.Definition
    if err := json.Unmarshal([]byte(def.DefinitionJSON), &wf); err != nil {
        return nil, err
    }
    for i := range wf.Nodes {
        if wf.Nodes[i].Code == inst.CurrentNode {
            return &wf.Nodes[i], nil
        }
    }
    return nil, fmt.Errorf("node %s not found", inst.CurrentNode)
}

func (s *RBACService) CheckPermission(ctx context.Context, userID uint, action rbac.Action, entityID uint) rbac.Decision {
    // 加载实体
    var entity pmocker.PMEntity
    if err := global.GVA_DB.WithContext(ctx).First(&entity, entityID).Error; err != nil {
        return rbac.Decision{Allowed: false, Reason: "entity not found"}
    }

    // ① gva RBAC（简化：v1 直接放行，由 gva casbin 中间件负责 API 级权限）
    // M2 阶段先用 gva 现有 casbin，PMocker 的细粒度权限在 M3 完善

    // ② EPS 节点权限（v1 简化：PMO 管理员/项目经理可访问所有，成员只能看自己参与的项目）
    // M3 阶段补充 EPS 树权限继承

    // ③ 实体状态权限
    allowed, reason := s.checkEntityState(entity.Status, action, entity.OwnerID, userID)
    if !allowed {
        return rbac.Decision{Allowed: false, Reason: reason}
    }

    return rbac.Decision{Allowed: true}
}

// checkEntityState 实体状态权限检查
func (s *RBACService) checkEntityState(status string, action rbac.Action, ownerID *uint, userID uint) (bool, string) {
    switch status {
    case "draft":
        // 草稿态：仅创建者可改
        if action == rbac.ActionUpdate || action == rbac.ActionDelete {
            if ownerID == nil || *ownerID != userID {
                return false, "draft state: only owner can modify"
            }
        }
    case "reviewing":
        // 评审态：仅审核人可批注，其他人只读
        if action == rbac.ActionUpdate {
            return false, "reviewing state: only approvers can annotate"
        }
    case "published", "released":
        // 发布态：只读，修改走变更管理
        if action == rbac.ActionUpdate || action == rbac.ActionDelete {
            return false, "published state: read-only, changes via change management"
        }
    case "archived":
        // 归档态：仅 PMO 可恢复
        if action == rbac.ActionUpdate {
            return false, "archived state: only PMO can restore"
        }
    }
    return true, ""
}

    // 初始化 DB（使用测试 SQLite）
    global.GVA_DB = initialize.Gorm()
    initialize.RegisterPMockerTables()

    ctx := context.Background()
    svc := &EAVService{}

    // 注册 schema
    err := svc.RegisterEntityType(ctx, eavtypes.EntityType{

        ModuleCode: "requirement",
        Name:       "测试需求",
    })

        FieldLabel:  "优先级",
        DataType:    eavtypes.DataTypeEnum,
        OptionsJSON: `["P0","P1","P2","P3"]`,
    })
    if err != nil {
        t.Fatalf("RegisterFieldDef failed: %v", err)
    }

    // 创建实体

        EntityType: "test_requirement",
        Title:      "测试需求1",
        Status:     "draft",
        Attrs: map[string]interface{}{

        },

    entity, err := svc.GetEntity(ctx, id)

    if entity.Attrs["priority"] != "P1" {

    // 清理
    svc.DeleteEntity(ctx, id)
}
