# M8 PMocker MCP 集成 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 AI 编辑器通过 MCP 调用 PMocker 的项目管理能力（8 个内置工具 + 30+ 动态工具），支持开发模式和实例模式。

**Architecture:** 复用 GVA MCP 独立进程（端口 8889 开发 / 8899 实例）。PMocker 内置工具实现 `McpTool` 接口放在 `gva/server/mcp/pmocker_*.go`；动态工具通过 9 模块 `seed.yaml` 预定义，AutoInitIfEmpty 自动插入 `sys_mcp_tools` 表；实例模式由 `pmocker run` 拉起 gva-server + MCP 双进程，自动签发长期 API Token。

**Tech Stack:** Go 1.22+, mark3labs/mcp-go, gin-vue-admin v3.0.0, SQLite, Cobra CLI

## Global Constraints

- 工具文件统一放在 `gva/server/mcp/` 目录，包名 `mcpTool`
- 每个 `pmocker_*` 工具在 `init()` 中调用 `RegisterTool`，重名会 panic（启动期暴露冲突）
- 上游调用用 `postUpstream[T](ctx, "/api/pmocker/<mod>/<action>", body)` 或 `getUpstream[T](ctx, endpoint, query)`，路径必须含 `/api` 前缀
- 参数从 `request.GetArguments()` 提取，类型断言 `args["name"].(string)`
- 结果用 `textResultWithJSON(prefix, data)` 返回
- MCP 端口：开发模式 8889（已有），实例模式 8899（固定，用户已确认）
- 鉴权 token 有效期 -1（100 年），通过 API Token 接口签发

```go
package mcpTool

import (
	"context"
	"encoding/json"
	"net/http"
	"net/http/httptest"
	"testing"

	"github.com/flipped-aurora/gin-vue-admin/server/global"
	"github.com/flipped-aurora/gin-vue-admin/server/model/system"
	"github.com/spf13/viper"
)

// mockUpstreamServer 启动一个 mock HTTP 服务模拟 GVA 后端，返回预设 JSON 响应。
// 自动设置 global.GVA_CONFIG.MCP.UpstreamBaseURL 指向 mock 服务。
func mockUpstreamServer(t *testing.T, path string, response any) *httptest.Server {
	t.Helper()
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.URL.Path != path {
			w.WriteHeader(http.StatusNotFound)
			return
		}
		w.Header().Set("Content-Type", "application/json")
		json.NewEncoder(w).Encode(response)
	}))
	t.Cleanup(server.Close)
	// 设置全局配置指向 mock 服务
	if global.GVA_VP == nil {
		global.GVA_VP = viper.New()
	}
	global.GVA_CONFIG.MCP.UpstreamBaseURL = server.URL
	return server
}

// mockMCPContext 返回带测试 token 的 context，用于 postUpstream 鉴权。
func mockMCPContext() context.Context {
	ctx := context.Background()
	ctx = context.WithValue(ctx, "mcp_auth_token", "test-token")
	return ctx
}
```

```go
package mcpTool

import (
	"context"
	"fmt"

	"github.com/mark3labs/mcp-go/mcp"
)

func init() {
	RegisterTool(&RequirementCreator{})
}

type RequirementCreator struct{}

func (r *RequirementCreator) New() mcp.Tool {
	return mcp.NewTool("pmocker_create_requirement",
		mcp.WithDescription(`创建项目需求。

**参数说明：**
- title: 需求标题（必填）
- description: 需求详细描述
- priority: 优先级 (high/medium/low)，默认 medium
- module: 归属模块（可选，如 scope/schedule/cost）`),
		mcp.WithString("title",
			mcp.Description("需求标题"),
		),
		mcp.WithString("description",
			mcp.Description("需求详细描述"),
		),
		mcp.WithString("priority",
			mcp.Description("优先级 (high/medium/low)"),
			mcp.DefaultString("medium"),
		),
		mcp.WithString("module",
			mcp.Description("归属模块（可选）"),
		),
	)
}

func (r *RequirementCreator) Handle(ctx context.Context, request mcp.CallToolRequest) (*mcp.CallToolResult, error) {
	args := request.GetArguments()

	title, ok := args["title"].(string)
	if !ok || title == "" {
		return nil, fmt.Errorf("title 参数必填")
	}

	body := map[string]any{
		"title": title,
	}
	if desc, ok := args["description"].(string); ok && desc != "" {
		body["description"] = desc
	}
	priority := "medium"
	if p, ok := args["priority"].(string); ok && p != "" {
		priority = p
	}
	body["priority"] = priority
	if mod, ok := args["module"].(string); ok && mod != "" {
		body["module"] = mod
	}

	resp, err := postUpstream[map[string]any](ctx, "/api/pmocker/requirement/create", body)
	if err != nil {
		return nil, fmt.Errorf("创建需求失败: %w", err)
	}

	return textResultWithJSON("需求创建成功：", map[string]any{
		"success": true,
		"data":    resp.Data,
	})
}
```
